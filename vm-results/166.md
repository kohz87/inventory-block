## Blocker

### B1. A hidden missing skill stat changes the number of fight rounds

The new generic missing-stat refusal is longer than the visible-stat refusal (`resolve.js:125-129,155-161`). `fitCue` measures successful check lines with `plain: true`, but `cueText` passes refusal text through unchanged (`cue.js:68-85,98-107`). Thus the supposedly visibility-independent measurement still depends on whether the missing skill stat is hidden. Because fitting tries fewer rounds before shortening the cue, the longer refusal can discard a round. This conflicts with the lead's explicit forward rule that hiding a stat never changes how many rounds are told. The permitted guard/Re-read caused by changed refusal wording is a separate decision; this repro compares two fresh rolls at the reviewed head with the same plan and seed.

On the real host, the repro below uses valid Character, Skills and Foes blocks, a normal `dice: 2` payload, seed 33, three STR checks, one refused skill check, one STR act and one foe attack. The Character is missing DEX; the skill selects DEX. The only definition difference is `stat1.prompt = false` in the hidden case. The planner plans are equal, all three check results and the first four rounds' dice and outcomes are equal. With the stat visible, the five-round cue is 2,400 characters (600 tokens), and the fight records **5** rounds, stopping `decisive`; round five adds `cost` and `fail`. With it hidden, the five-round plain measurement is **2,416** characters (604 tokens), so the fight records only **4** rounds and stops `budget`. The narrator gets the appropriate public or generic refusal in each case. The witness may differ in its refusal text as explicitly allowed; the changed number of rounds is the blocker.

`round5b.test.js:172-186` varies budgets and visibility for rolled lines but explicitly uses `refused: []`, so it does not cover this interaction; the no-hidden-difference claim in `docs/v2/RPG.md:199` is not true for it. A fix needs a visibility-independent fitting measure for these refusals while still bounding the actual emitted text to the cue budget.

Run this at the requested head; it does not modify the checkout or restore any older implementation:

```sh
node --input-type=module <<'JS'
import assert from 'node:assert/strict';
import { createHost } from './testkit/index.js';
import { LEAN_RPG_TEMPLATES } from './src/v2/lean-rpg/templates.js';
import { cueText, cueTokens } from './src/v2/lean-rpg/cue.js';
import { success } from './tests/v2/data-modules/helpers.js';

const template = name => structuredClone(LEAN_RPG_TEMPLATES.find(x => x.name === name).definition);
const who = 'Hero' + 'x'.repeat(46), foe = 'Foe' + 'z'.repeat(46);
const answer = [
  'pace: all',
  `act ${who} | stat: STR | target: ${foe}`,
  `foe ${foe} | target: ${who} | defend: STR`,
  ...Array.from({ length: 3 }, () => `check ${who} | stat: STR | difficulty: normal`),
  `check ${who} | skill: Silent Trick | difficulty: normal`,
].join('\n');

const cases = [];
for (const hidden of [false, true]) {
  const host = await createHost({ callModel: async () => answer });
  try {
    host.features.set('story-block.rpg', true);
    await host.send('Begin.', 'Opening.');
    const hero = template('Character');
    hero.slug = 'hero'; hero.name = who; hero.state.who.max = 60; hero.state.who.default = who;
    hero.kinds = {}; hero.cards = []; delete hero.state.dex;
    success(await host.sdk.command('dm.define', { definition: hero }));
    const skills = template('Skills');
    skills.slug = 'skills'; skills.state.who.max = 60; skills.state.who.default = who; skills.kinds = {};
    skills.state.skill1.default = 'Silent Trick'; skills.state.stat1.default = 'dex';
    if (hidden) skills.state.stat1.prompt = false;
    success(await host.sdk.command('dm.define', { definition: skills }));
    const foes = template('Foes');
    foes.state.foe1.max = 60; foes.state.foe1.default = foe; foes.state.count1.default = 1;
    success(await host.sdk.command('dm.define', { definition: foes }));
    const run = host.bus.run;
    host.bus.run = (type, payload, options) => type === 'rpg.resolve'
      ? run(type, payload, { ...options, seed: 33 }) : run(type, payload, options);
    const sent = await host.send('We act.', 'A reply.');
    const record = host.ctx.chat.at(-1).extra.storySuite;
    const roll = record.receipts.find(r => r.type === 'rpg.resolve');
    const plan = record.spans.find(s => s.operation?.type === 'rpg.resolve').operation.payload.plan;
    assert.ok(roll); assert.equal(roll.seed, 33); assert.equal(roll.replan.payload.dice, 2);
    assert.match(JSON.stringify(sent.request), hidden ? /has no stat on the sheet for this skill/ : /has no DEX on the sheet/);
    cases.push({ hidden, roll, plan });
  } finally { host.close(); }
}
const [shown, hidden] = cases;
const a = shown.roll.data.v2.result, b = hidden.roll.data.v2.result;
assert.deepEqual(shown.plan, hidden.plan);
assert.deepEqual(a.checks, b.checks);
assert.equal(a.rounds.length, 5); assert.equal(a.stop, 'decisive');
assert.equal(b.rounds.length, 4); assert.equal(b.stop, 'budget');
assert.deepEqual(a.rounds.slice(0, 4), b.rounds);
const fiveHidden = cueText({ checks: b.checks, rounds: [...b.rounds, a.rounds[4]],
  refused: b.refused.map(x => x.text), decisive: true }, 0, { plain: true });
assert.equal(shown.roll.summary.length, 2400);
assert.equal(fiveHidden.length, 2416); assert.equal(cueTokens(fiveHidden), 604);
console.log(JSON.stringify(cases.map(({ hidden, roll }) => ({ hidden, seed: roll.seed,
  rounds: roll.data.v2.result.rounds.length, stop: roll.data.v2.result.stop,
  cueLength: roll.summary.length, refusal: roll.data.v2.result.refused[0].text }))));
JS
```
